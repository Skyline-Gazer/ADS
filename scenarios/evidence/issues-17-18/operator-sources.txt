    p = subprocess.run(args, cwd=cwd, env=env, text=True, capture_output=True, timeout=150)

def digest(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()

def globals_snapshot():

    paths += sorted((home / '.codex/rules').glob('*.rules'))

                     'mtime_ns': p.stat().st_mtime_ns, 'mode': p.stat().st_mode & 0o777}
            if p.is_file() else None for p in paths}
